# requirements-swarm-showcase.txt
# Python dependencies for swarm-showcase.py
#
# The showcase script auto-installs AI framework integration packages from
# the local integrations/ directory using `pip install -e`. This file lists
# the *runtime* Python packages those integrations depend on, so you can
# pre-install them before running the swarm (recommended for CI/offline use).
#
# Usage:
#   pip install -r scripts/requirements-swarm-showcase.txt
#
# Then install the local integration packages:
#   pip install -e integrations/langchain
#   pip install -e integrations/crewai
#   pip install -e integrations/llamaindex
#   pip install -e integrations/autogpt
#   pip install -e integrations/superagi
#   pip install -e integrations/agno
#   pip install -e integrations/camel-ai
#   pip install -e integrations/composio
#   pip install -e integrations/dify
#   pip install -e integrations/haystack
#   pip install -e integrations/ironclaw
#   pip install -e integrations/metagpt
#   pip install -e integrations/moltbook
#   pip install -e integrations/pydantic-ai
#   pip install -e integrations/semantic-kernel
#
# Note: agent-zero uses no-package imports (numbered tool files loaded via
# importlib). raw-http uses only the `requests` package (auto-installed).
#
# Node.js integrations (mastra, flowise, openclaw, vercel-ai, openai-functions,
# n8n, gemini) require Node.js ≥18 and npm. The script will auto-build them
# via `npm install && npm run build` in each integration directory.
